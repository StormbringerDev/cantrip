cmake_minimum_required(VERSION 3.16)
project(Cantrip VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/cmake")

option(CANTRIP_BUILD_CLI "Build the CLI entry point for Cantrip" OFF)
option(CANTRIP_BUILD_TESTS "Build unit tests for Cantrip" OFF)
option(CANTRIP_WARN_AS_ERROR "Treat warnings as errors." OFF)

if(CANTRIP_BUILD_TESTS)
  enable_testing()
  add_subdirectory(external/Unity)
  add_subdirectory(tests)
endif()
